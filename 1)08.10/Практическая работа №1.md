# Практическая раота №1

## Подготовка базы данных в Microsoft SQL Server

Откройте среду **SQL Server Management Studio (SSMS)**, **Azure Data Studio** или расширение **MSSQL** в VS Code и выполните T-SQL скрипт:

```sql
-- Создание новой базы данных для практической работы
CREATE DATABASE ShopPracticeDb;
GO

USE ShopPracticeDb;
GO

-- Создание таблицы товаров
CREATE TABLE Products (
    Id INT IDENTITY(1, 1) PRIMARY KEY,
    Title NVARCHAR(100) NOT NULL,
    Price DECIMAL(10, 2) NOT NULL,
    InStock BIT NOT NULL DEFAULT 1,
    CreatedAt DATETIME2 DEFAULT SYSDATETIME()
);
GO
```

## Создание консольного проекта .NET

Откройте терминал (PowerShell или bash) и последовательно выполните команды:

```bash
# 1. Создание нового консольного приложения
dotnet new console -n AdoNetPractice

# 2. Переход в папку проекта
cd AdoNetPractice

# 3. Добавление официального пакета ADO.NET для работы с SQL Server
dotnet add package Microsoft.Data.SqlClient
```

## Разбор строки подключения

Строка подключения определяет параметры авторизации и адрес сервера:

- **Windows-аутентификация (рекомендуется для локальной разработки):**
  `Server=localhost;Database=ShopPracticeDb;Integrated Security=True;TrustServerCertificate=True;`
- **SQL-аутентификация (логин и пароль):**
  `Server=localhost;Database=ShopPracticeDb;User Id=sa;Password=YourStrongPassword!;TrustServerCertificate=True;`

> **Важно:** Параметр `TrustServerCertificate=True` необходим для локальных сред разработки, чтобы клиент доверял самоподписанному SSL/TLS-сертификату сервера.

## Полный исходный код программы (`Program.cs`)

Замените все содержимое файла `Program.cs` следующим кодом:

```csharp
using System;
using System.Data;
using Microsoft.Data.SqlClient;

class Program
{
    // Укажите имя вашего локального сервера, если оно отличается от localhost
    private const string ConnectionString =
        "Server=localhost;Database=ShopPracticeDb;Integrated Security=True;TrustServerCertificate=True;";

    static void Main()
    {
        Console.WriteLine("=== Старт практической работы ADO.NET ===");

        try
        {
            // 1. Проверка доступности подключения
            CheckConnection();

            // 2. Добавление записей в таблицу (ExecuteNonQuery)
            Console.WriteLine("\n--- Вставка данных ---");
            InsertProduct("Игровая мышь", 3200.50m, true);
            InsertProduct("Механическая клавиатура", 6500.00m, true);
            InsertProduct("Коврик для мыши", 850.00m, false);

            // 3. Вычисление агрегатного значения (ExecuteScalar)
            Console.WriteLine("\n--- Подсчет записей ---");
            int totalCount = GetTotalProductsCount();
            Console.WriteLine($"Всего товаров в базе данных: {totalCount}");

            // 4. Потоковое чтение строк (ExecuteReader)
            Console.WriteLine("\n--- Выборка товаров ---");
            ReadAllProducts();
        }
        catch (SqlException ex)
        {
            Console.WriteLine($"[Ошибка базы данных]: {ex.Message}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"[Непредвиденная ошибка]: {ex.Message}");
        }

        Console.WriteLine("\n=== Практическая работа завершена ===");
    }

    /// <summary>
    /// Проверка открытия и закрытия физического подключения.
    /// </summary>
    static void CheckConnection()
    {
        using (SqlConnection connection = new SqlConnection(ConnectionString))
        {
            connection.Open();
            Console.WriteLine($"Подключение успешно открыто. Состояние: {connection.State}");
        }
    }

    /// <summary>
    /// Вставка строки с безопасной передачей параметров через SqlCommand.
    /// </summary>
    static void InsertProduct(string title, decimal price, bool inStock)
    {
        string query = @"INSERT INTO Products (Title, Price, InStock)
                         VALUES (@Title, @Price, @InStock);";

        using (SqlConnection connection = new SqlConnection(ConnectionString))
        {
            connection.Open();

            using (SqlCommand command = new SqlCommand(query, connection))
            {
                // Привязка параметров с явным указанием типов данных
                command.Parameters.Add("@Title", SqlDbType.NVarChar, 100).Value = title;
                command.Parameters.Add("@Price", SqlDbType.Decimal).Value = price;
                command.Parameters.Add("@InStock", SqlDbType.Bit).Value = inStock;

                int rowsAffected = command.ExecuteNonQuery();
                Console.WriteLine($"Добавлен товар: {title} (затронуто строк: {rowsAffected})");
            }
        }
    }

    /// <summary>
    /// Получение одного значения с помощью ExecuteScalar.
    /// </summary>
    static int GetTotalProductsCount()
    {
        string query = "SELECT COUNT(*) FROM Products;";

        using (SqlConnection connection = new SqlConnection(ConnectionString))
        {
            connection.Open();

            using (SqlCommand command = new SqlCommand(query, connection))
            {
                object result = command.ExecuteScalar();
                return Convert.ToInt32(result);
            }
        }
    }

    /// <summary>
    /// Потоковое чтение данных через SqlDataReader.
    /// </summary>
    static void ReadAllProducts()
    {
        string query = "SELECT Id, Title, Price, InStock FROM Products ORDER BY Id ASC;";

        using (SqlConnection connection = new SqlConnection(ConnectionString))
        {
            connection.Open();

            using (SqlCommand command = new SqlCommand(query, connection))
            using (SqlDataReader reader = command.ExecuteReader())
            {
                Console.WriteLine("-----------------------------------------------------------------");
                Console.WriteLine($"{"ID",-4} | {"Наименование",-25} | {"Цена",-10} | {"Статус",-12}");
                Console.WriteLine("-----------------------------------------------------------------");

                while (reader.Read())
                {
                    int id = reader.GetInt32(0);
                    string title = reader.GetString(1);
                    decimal price = reader.GetDecimal(2);
                    bool inStock = reader.GetBoolean(3);

                    string status = inStock ? "В наличии" : "Нет на складе";
                    Console.WriteLine($"{id,-4} | {title,-25} | {price,10:C} | {status,-12}");
                }

                Console.WriteLine("-----------------------------------------------------------------");
            }
        }
    }
}
```
